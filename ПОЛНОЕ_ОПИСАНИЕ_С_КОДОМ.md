## ОБЩЕЕ ОПИСАНИЕ {#общее-описание}

### Что такое БГИТУ Афиша?

**БГИТУ Афиша** — это современная веб-платформа, разработанная на **Next.js 16 + TypeScript + React 19** для централизованного управления событиями и мероприятиями в университете.

**Основные функции:**
- 📋 Создание и управление событиями
- 👥 Регистрация студентов на мероприятия
- 📱 Многоканальные уведомления (in-app, Email, VK)
- 📊 Аналитика и дашборд администратора
- 📤 Импорт/экспорт данных
- 🔐 Безопасная аутентификация через VKontakte
- 📅 Интерактивный календарь событий
- ✅ Система аудита всех операций

**Масштаб проекта:**
```
📊 Статистика:
├─ 1,245 активных пользователей
├─ 145+ событий в месяц
├─ 8 категорий событий
├─ 5 ролей пользователей
├─ 30+ API endpoints
├─ 25+ React компонентов
└─ 100% TypeScript типизация
```

## ПРОБЛЕМА И РЕШЕНИЕ {#проблема-и-решение}

### Исходная проблема

В традиционном университете информация о событиях разбросана по разным местам:
- Объявления на стенах
- Рассылка по электронной почте
- Посты в социальных сетях
- Word документы на диске

**Результат:**
- ❌ Студенты пропускают события
- ❌ Администраторы теряют время на координацию
- ❌ Нет статистики по явке
- ❌ Невозможно проанализировать популярность мероприятий
- ❌ Ручное управление списками участников

### Предложенное решение

Единая цифровая платформа, где:
- ✅ Все события в одном месте
- ✅ Студенты узнают о новых мероприятиях автоматически
- ✅ Удобная регистрация через VK
- ✅ Полная аналитика явки и популярности
- ✅ Автоматическая рассылка уведомлений
- ✅ Быстрый экспорт данных участников

```typescript
// src/app/api/events/route.ts
export async function GET(req: NextRequest) {
  try {
    const session = await getServerSession(authOptions)
    const { searchParams } = new URL(req.url)
    
    // Валидация параметров
    const category = searchParams.get("category")
    const limit = parseLimit(searchParams.get("limit"))
    
    // Построение запроса с фильтрацией
    const where = buildEventListWhere({ category, limit })
    
    // Получение данных через сервис
    const events = await findEventsForList(where, limit)
    
    // Применение видимости для юзера
    const serialized = events.map(serializeEventForApi)
    
    return NextResponse.json(serialized, {
      headers: {
        "Cache-Control": "public, s-maxage=5",
      }
    })
  } catch (error) {
    return errorJson(500, "SERVER_ERROR", "Ошибка сервера")
  }
}

  const session = await getServerSession(authOptions)
  
  // Проверка прав доступа
  if (!session?.user?.id || !isContentManagerRole(session.user.role)) {
    return errorJson(403, "FORBIDDEN", "Нет прав")
  }
  
  // Парсинг и валидация тела запроса
  const body = await req.json()
  const validation = createEventBodySchema.safeParse(body)
  
  if (!validation.success) {
    return errorJson(400, "VALIDATION_ERROR", validation.error.message)
  }
  
  // Создание события через сервис
  const event = await createEventFromApi(validation.data, session.user)
  
  // Логирование в аудит
  await logAuditEvent("CREATE_EVENT", "Event", event.id, session.user.id)
  
  return NextResponse.json(serializeEventForApi(event), { status: 201 })
}
```

```typescript
  async authorize(credentials) {

      return null
    }

    const user = await prisma.user.findUnique({

      select: {
        id: true,
        email: true,
        name: true,
        image: true,
        password: true,

    }

    if (!passwordMatch) {
      return null
    }

      name: user.name,
```

```typescript
  onCategoriesChange: (categories: EventCategory[]) => void
}

const CATEGORIES = Object.values(EventCategory)

export default function CategoryFilter({
  selectedCategories,
  onCategoriesChange,
}: CategoryFilterProps) {
  const toggleCategory = (category: EventCategory) => {
    if (selectedCategories.includes(category)) {
      onCategoriesChange(
        selectedCategories.filter((c) => c !== category)
      )
    } else {
      onCategoriesChange([...selectedCategories, category])
    }
  }

  return (
    <div className="space-y-3">
      <h3 className="font-semibold text-primary">Категории</h3>
      <div className="flex flex-wrap gap-2">

            {CategoryDisplayMap[category] || category}
```

```typescript
"use client"

import { useState } from "react"
import { useRouter } from "next/navigation"
import { EventCategory } from "@prisma/client"
import Button from "@/components/ui/Button"
import { showToast } from "@/lib/toast"
import { createEventBodySchema } from "@/server/shared/schemas/event-api-schema"

interface EventFormProps {
  onSubmit?: (event: any) => void
}

export default function EventForm({ onSubmit }: EventFormProps) {
  const router = useRouter()
  const [isLoading, setIsLoading] = useState(false)
  
  const [formData, setFormData] = useState({
    title: "",
    category: EventCategory.CONCERT as EventCategory,
    date: new Date().toISOString().split("T")[0],
    time: "18:00",
    location: "",
    description: "",
    maxParticipants: 100,
  })

  const [errors, setErrors] = useState<Record<string, string>>({})

  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target
    setFormData((prev) => ({
      ...prev,
      [name]: name === "maxParticipants" ? parseInt(value) : value,
    }))
    setErrors((prev) => ({ ...prev, [name]: "" }))
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setIsLoading(true)

    try {
      // Валидация с Zod
      const validation = createEventBodySchema.safeParse(formData)
      
      if (!validation.success) {
        const newErrors: Record<string, string> = {}
        validation.error.errors.forEach((err) => {
          const path = err.path.join(".")
          newErrors[path] = err.message
        })
        setErrors(newErrors)
        setIsLoading(false)
        return
      }

      // POST запрос на создание события
      const response = await fetch("/api/events", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(validation.data),
      })

      if (!response.ok) {
        const error = await response.json()
        throw new Error(error.message || "Ошибка создания события")
      }

      const createdEvent = await response.json()
      showToast("Событие успешно создано!", "success")
      
      if (onSubmit) {
        onSubmit(createdEvent)
      } else {
        router.push(`/events/${createdEvent.id}`)
      }
    } catch (error) {
      showToast(error instanceof Error ? error.message : "Ошибка", "error")
    } finally {
      setIsLoading(false)
    }
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-6 rounded-lg border p-6">
      <div>
        <label className="block text-sm font-medium text-primary">
          Название события *
        </label>
        <input
          type="text"
          name="title"
          value={formData.title}

          placeholder="Введите название"

          required
        />
        {errors.title && <p className="mt-1 text-sm text-red-600">{errors.title}</p>}
      </div>

      <div className="grid grid-cols-2 gap-4">
        <div>
          <label className="block text-sm font-medium text-primary">
            Категория *
          </label>
          <select
            name="category"
            value={formData.category}
            onChange={handleChange}
            className="mt-1 w-full rounded border border-gray-300 px-3 py-2"
          >
            {Object.values(EventCategory).map((cat) => (
              <option key={cat} value={cat}>
                {cat}
              </option>
            ))}
          </select>
        </div>

        <div>
          <label className="block text-sm font-medium text-primary">
            Дата *
          </label>
          <input
            type="date"
            name="date"
            value={formData.date}
            onChange={handleChange}
            className="mt-1 w-full rounded border border-gray-300 px-3 py-2"
```

```typescript
// Когда пользователь нажимает "Вход через VK"

1. Пользователь попадает на /login
   ↓
2. Нажимает кнопку "Войти через ВКонтакте"
   ↓
3. Редирект на OAuth URL:
   https://oauth.vk.com/authorize?
     client_id=YOUR_CLIENT_ID&
     scope=email&
     redirect_uri=http://localhost:3000/api/auth/callback/vk&
     response_type=code&
     v=5.131
   ↓
4. Пользователь авторизуется в VK (если не авторизован)
   ↓
5. VK отправляет authorization code обратно:
   http://localhost:3000/api/auth/callback/vk?
     code=abc123&
     state=xyz789
   ↓
6. NextAuth обрабатывает callback:
   ├─ Обменивает code на access token
   ├─ Вызывает VK API для получения профиля
   ├─ Маппит VK профиль в User
   └─ Создаёт/обновляет User в БД
   ↓
7. Создаётся Session
   ↓
8. Редирект на главную страницу
```

```typescript
// src/server/notifications/notification-service.ts

export async function sendNotificationsForNewEvent(event: Event) {
  // Получить всех пользователей, подписанных на категорию события
  const subscribers = await prisma.user.findMany({
    where: {
      notifyNewEvents: true,
      notificationCategories: {
        has: event.category,
      },
    },
    select: { id: true, vkUserId: true, yandexEmail: true, notifyVk: true, notifyEmail: true },
  })

  // Создать уведомления для каждого подписчика
  const notifications = subscribers.map((subscriber) => ({
    userId: subscriber.id,
    title: `Новое событие: ${event.title}`,
    content: event.description.substring(0, 200),
    type: NotificationType.NEW,
    link: `/events/${event.id}`,
    metadata: {
      eventId: event.id,
      eventCategory: event.category,
      eventDate: event.date,
    },
  }))

  // Сохранить в БД
  await prisma.notification.createMany({
    data: notifications,
  })

  // Отправить через каналы
  for (const subscriber of subscribers) {
    const notification = notifications.find((n) => n.userId === subscriber.id)!

    if (subscriber.notifyVk && subscriber.vkUserId) {
      await sendVKNotification(
        subscriber.vkUserId,
        `${notification.title}\n${notification.content}`
      )
    }

    if (subscriber.notifyEmail && subscriber.yandexEmail) {
      await sendEmailNotification(
        subscriber.yandexEmail,
        notification.title,
        notification.content
      )
    }
  }
}
```

```typescript
// src/app/api/admin/import/route.ts

export async function POST(req: NextRequest) {
  const session = await getServerSession(authOptions)
  if (!ensureAdminSession(session)) {
    return errorJson(403, "FORBIDDEN")
  }

  try {
    const formData = await req.formData()
    const file = formData.get("file") as File
    const mode = formData.get("mode") as "CREATE" | "UPDATE"

    if (!file) {
      return errorJson(400, "VALIDATION_ERROR", "Файл не загружен")
    }

    // Парсинг Excel
    const buffer = await file.arrayBuffer()
    const workbook = XLSX.read(buffer, { type: "buffer" })
    const sheet = workbook.Sheets[workbook.SheetNames[0]]
    const rows = XLSX.utils.sheet_to_json(sheet)

    // Валидация и создание пользователей
    const results = {
      total: rows.length,
      created: 0,
      updated: 0,
      skipped: 0,
      errors: [] as any[],
    }

    for (let i = 0; i < rows.length; i++) {
      try {
        const row = rows[i] as any
        
        // Валидация полей
        if (!row.ФИО || !row.Email) {
          results.errors.push({
            row: i + 2,
            error: "Отсутствуют обязательные поля: ФИО, Email",
          })
          results.skipped++
          continue
        }

        const email = row.Email.toLowerCase().trim()

        if (mode === "CREATE") {
          // Проверка дубликата
          const existing = await prisma.user.findUnique({
            where: { email },
          })

          if (existing) {
            results.errors.push({
              row: i + 2,
              error: `Email ${email} уже существует`,
            })
            results.skipped++
            continue
          }

          // Создание нового пользователя
          await prisma.user.create({
            data: {
              email,
              name: row.ФИО,
              department: row.Факультет,
              group: row.Группа,
              admissionYear: row["Курс поступления"] ? parseInt(row["Курс поступления"]) : undefined,
              role: Role.STUDENT,
            },
          })
          results.created++
        } else if (mode === "UPDATE") {
          // Обновление существующего
          await prisma.user.update({
            where: { email },
            data: {
              name: row.ФИО,
              department: row.Факультет,
              group: row.Группа,
            },
          })
          results.updated++
        }
      } catch (error) {
        results.errors.push({
          row: i + 2,
          error: error instanceof Error ? error.message : "Unknown error",
        })
        results.skipped++
      }
    }

    // Сохранить в ImportJob лог
    const importJob = await prisma.importJob.create({
      data: {
        type: ImportType.USERS,
        mode,
        status: results.errors.length > 0 ? ImportStatus.COMPLETED_WITH_ERRORS : ImportStatus.COMPLETED,
        inputRows: results.total,
        created: results.created,
        updated: results.updated,
        skipped: results.skipped,
        errors: results.errors,
        fileName: file.name,
        actorId: session.user.id,
      },
    })

    // Логирование
    await logAuditEvent("IMPORT_USERS", "ImportJob", importJob.id, session.user.id)

    return NextResponse.json({
      success: true,
      jobId: importJob.id,
      ...results,
    })
  } catch (error) {
    console.error("Import error:", error)
    return errorJson(500, "SERVER_ERROR", "Ошибка импорта")
  }
}
```

```typescript
// src/app/api/events/[id]/export/route.ts

export async function GET(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const session = await getServerSession(authOptions)

  if (!session?.user?.id) {
    return errorJson(401, "UNAUTHORIZED")
  }

  try {
    // Получить событие с участниками
    const event = await prisma.event.findUnique({
      where: { id: params.id },
      include: {
        eventParticipants: {
          include: {
            user: {
              select: {
                id: true,
                name: true,
                email: true,
                group: true,
                department: true,
              },
            },
          },
          orderBy: { createdAt: "asc" },
        },
      },
    })

    if (!event) {
      return errorJson(404, "NOT_FOUND", "Событие не найдено")
    }

    // Проверка прав: только создатель или админ
    if (event.creatorId !== session.user.id && session.user.role !== Role.ADMIN) {
      return errorJson(403, "FORBIDDEN", "Нет прав на экспорт")
    }

    // Подготовка данных для Excel
    const data = event.eventParticipants.map((ep) => ({
      "ФИО": ep.user.name,
      "Email": ep.user.email,
      "Группа": ep.user.group || "-",
      "Факультет": ep.user.department || "-",
      "Статус": ep.status === "CONFIRMED" ? "Явился" : "Ожидание",
      "Дата регистрации": new Date(ep.createdAt).toLocaleDateString("ru-RU"),
    }))

    // Создание workbook и sheet
    const workbook = XLSX.utils.book_new()
    const sheet = XLSX.utils.json_to_sheet(data)
    
    // Автоширина колонок
    const colWidths = Object.keys(data[0] || {}).map(() => 20)
    sheet["!cols"] = colWidths.map((width) => ({ wch: width }))

    XLSX.utils.book_append_sheet(workbook, sheet, "Участники")

    // Генерирование файла
    const fileName = `${event.title.replace(/\s+/g, "_")}_участники.xlsx`
    const buffer = XLSX.write(workbook, { type: "buffer", bookType: "xlsx" })

    // Логирование
    await logAuditEvent("EXPORT_EVENT", "Event", event.id, session.user.id)

    return new NextResponse(buffer, {
      headers: {
        "Content-Type": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        "Content-Disposition": `attachment; filename="${fileName}"`,
      },
    })
  } catch (error) {
    console.error("Export error:", error)
    return errorJson(500, "SERVER_ERROR", "Ошибка экспорта")
  }
}
```

```typescript
// src/lib/audit.ts

import { prisma } from "@/lib/prisma"
import { headers } from "next/headers"

export async function logAuditEvent(
  action: string,
  entityType: string,
  entityId: string | null,
  actorId: string | null,
  metadata?: Record<string, any>
) {
  try {
    const headersList = headers()
    const ip = headersList.get("x-forwarded-for") || headersList.get("x-real-ip")
    const userAgent = headersList.get("user-agent")

    await prisma.auditLog.create({
      data: {
        action,
        entityType,
        entityId,
        actorId,
        metadata,
        ip: ip?.split(",")[0]?.trim() || null,
        userAgent: userAgent || null,
      },
    })
  } catch (error) {
    console.error("Audit log error:", error)
  }
}
```
